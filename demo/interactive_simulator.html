#include <WiFi.h>
#include <LittleFS.h>

enum Priority : uint8_t { P0_CRIT = 0, P1_WARN = 1, P2_LOG = 2 };
enum Policy : uint8_t { POL_PRIORITY_PREEMPT = 0, POL_EDF_PRIO = 1, POL_FIFO = 2 };
enum HandshakeState : uint8_t {
  HS_BEACON_SCAN, HS_HELLO_REQ, HS_CAP_OFFER, HS_SCHED_ACK, HS_DATA_BURST, HS_FLASH_COMMIT
};

struct SensorPacket {
  uint32_t id;
  uint8_t  nodeId;
  uint8_t  priority;      // 0 = highest priority
  uint16_t sizeBytes;
  uint32_t expiryMs;      // deadline timestamp (ms)
  uint32_t crc32;
  char     payload[64];
};

constexpr size_t MAX_FLASH_SLOTS = 32;
SensorPacket muleBuffer[MAX_FLASH_SLOTS];
size_t usedSlots = 0;
Policy activePolicy = POL_PRIORITY_PREEMPT;

int comparePackets(const void* a, const void* b) {
  const SensorPacket* pa = (const SensorPacket*)a;
  const SensorPacket* pb = (const SensorPacket*)b;
  if (activePolicy == POL_FIFO) {
    return (pa->id > pb->id) - (pa->id < pb->id);
  }
  if (activePolicy == POL_EDF_PRIO) {
    // Lower score = higher urgency (combines priority and remaining TTL)
    uint32_t now = millis();
    int32_t remA = (int32_t)(pa->expiryMs - now);
    int32_t remB = (int32_t)(pb->expiryMs - now);
    int32_t scoreA = pa->priority * 10000 + remA;
    int32_t scoreB = pb->priority * 10000 + remB;
    return (scoreA > scoreB) - (scoreA < scoreB);
  }
  // POL_PRIORITY_PREEMPT: P0 (0) before P1 (1) before P2 (2), then older packet first
  if (pa->priority != pb->priority) return pa->priority - pb->priority;
  return (pa->id > pb->id) - (pa->id < pb->id);
}

bool storePacketToFlash(const SensorPacket& pkt) {
  if (usedSlots >= MAX_FLASH_SLOTS) {
    if (activePolicy == POL_FIFO) {
      // Evict oldest packet (index 0)
      memmove(&muleBuffer[0], &muleBuffer[1], (MAX_FLASH_SLOTS - 1) * sizeof(SensorPacket));
      usedSlots--;
    } else {
      // Find lowest-value victim slot to evict
      int victimIdx = -1;
      for (size_t i = 0; i < MAX_FLASH_SLOTS; i++) {
        if (muleBuffer[i].priority > pkt.priority) {
          if (victimIdx == -1 || muleBuffer[i].priority > muleBuffer[victimIdx].priority) {
            victimIdx = (int)i;
          }
        }
      }
      if (victimIdx == -1) return false; // Incoming packet is lower/equal priority than all buffered
      memmove(&muleBuffer[victimIdx], &muleBuffer[victimIdx + 1],
              (MAX_FLASH_SLOTS - 1 - victimIdx) * sizeof(SensorPacket));
      usedSlots--;
    }
  }
  muleBuffer[usedSlots++] = pkt;
  qsort(muleBuffer, usedSlots, sizeof(SensorPacket), comparePackets);

  File f = LittleFS.open("/dtn_queue.bin", FILE_WRITE);
  if (f) {
    f.write((uint8_t*)muleBuffer, usedSlots * sizeof(SensorPacket));
    f.close();
  }
  return true;
}
